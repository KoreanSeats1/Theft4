cmake_minimum_required(VERSION 3.24)
project(theft4_renderer_validation LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "../.." ABSOLUTE)
set(NATIVE "${ROOT}/glue/rexglue-sdk-main/src/graphics/gta4_native")
option(RENDERER_SANITIZERS "Enable address and undefined-behavior checks" OFF)
foreach(name masked_constant_replay_test frame_preparation_benchmark snapshot_pages_test cached_tuple_reuse_test)
  add_executable(${name} "${ROOT}/ios/tests/${name}.cpp")
  target_include_directories(${name} PRIVATE "${NATIVE}" "${ROOT}/glue/rexglue-sdk-main/thirdparty/xxHash")
  if(RENDERER_SANITIZERS)
    target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${name} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_test(NAME masked_replay COMMAND masked_constant_replay_test)
add_test(NAME frame_preparation COMMAND frame_preparation_benchmark)
add_test(NAME snapshot_pages COMMAND snapshot_pages_test)
add_test(NAME cached_tuples COMMAND cached_tuple_reuse_test)
option(RENDERER_CORPUS_AUDIT "Audit all embedded shader modules" ON)
if(RENDERER_CORPUS_AUDIT)
  set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
  add_subdirectory("${ROOT}/tools/XenosRecomp/thirdparty/zstd/build/cmake" zstd)
  add_executable(constant_projection_corpus "${ROOT}/ios/tests/constant_projection_corpus.cpp"
    "${ROOT}/tools/XenosRecomp/thirdparty/smol-v/source/smolv.cpp"
    "${ROOT}/LibertyRecompLib/shader/shader_cache.cpp")
  target_include_directories(constant_projection_corpus PRIVATE "${NATIVE}"
    "${ROOT}/LibertyRecompLib/shader" "${ROOT}/tools/XenosRecomp/thirdparty/smol-v/source"
    "${ROOT}/tools/XenosRecomp/thirdparty/zstd/lib"
    "${ROOT}/glue/rexglue-sdk-main/thirdparty/spirv-headers/include")
  target_compile_options(constant_projection_corpus PRIVATE -UNDEBUG)
  target_link_libraries(constant_projection_corpus PRIVATE libzstd_static)
  add_test(NAME shader_corpus COMMAND constant_projection_corpus)
endif()
